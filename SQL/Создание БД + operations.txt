Create table Login_password (
id_login Integer Primary key autoincrement,
login Varchar,
password varchar,
created_at DATE,
last_login datetime
)

Create table Bank_cards (
card_id Integer Primary key autoincrement,
card_number integer,
type_card varchar,
created_at Date,
expiration_date DATE 
)

Create table Users_messages (
message_id Integer Primary key autoincrement,
topic Varchar,
text_message text,
created_at Datetime
)

Create table Users (
user_id Integer Primary key autoincrement,
first_name Varchar,
last_name Varchar,
age integer,
phone_number varchar,
email varchar,
id_login integer,
bank_card integer,
user_message integer,
Foreign key (id_login) references Login_password (id_login),
foreign key (bank_card) references Bank_cards (card_id),
foreign key (user_message) references Users_messages (message_id)
)

SELECT 
    Users.first_name,
    Users.last_name,
    Login_password.login,
    Login_password.password,
    Login_password.created_at
FROM Users
INNER JOIN Login_password 
ON Users.id_login = Login_password.id_login

SELECT 
    Users.first_name,
    Users.last_name,
    Login_password.login,
    Login_password.password
FROM Users
INNER JOIN Login_password 
ON Users.id_login = Login_password.id_login

SELECT 
    Users.first_name,
    Users.last_name,
    Bank_cards.card_number,
    Bank_cards.type_card,
    Bank_cards.expiration_date,
    Users.age
FROM Users
INNER JOIN Bank_cards
ON Users.bank_card = Bank_cards.card_id
WHERE Users.age > 30

SELECT 
    Users.first_name,
    Users.last_name,
    Users_messages.topic,
    Users_messages.text_message,
    Users_messages.created_at
FROM Users
INNER JOIN Users_messages
ON Users.user_message = Users_messages.message_id
ORDER BY Users_messages.created_at
